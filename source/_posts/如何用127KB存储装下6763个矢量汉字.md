单片机上显示汉字，传统做法是烧一份点阵字库进 Flash。然而点阵的体积随字号平方增长，而且每种字号都要单独存一份。16×16 一份 211KB，再加 24×24 就是 687KB，再加 32×32 直接 1.5MB。你的 UI 想要三个字号？Flash 先哭了。

这篇文章是我对这个问题的回答：把 GB 2312 一二级全部 6763 个汉字存成笔画骨架，解码时在任意分辨率上重建。一份 127KB 的位流，服务所有字号，包括连续缩放。

字形几何来自 [GlyphWiki](https://glyphwiki.org) 的 KAGE 数据。KAGE 的表示方式天然适合这件事：每个字不是一堆轮廓曲线，而是"笔画 + 部件引用"——「湖」引用「氵」「古」「月」，每个引用带一个放置框。汉字本来就是搭积木搭出来的，KAGE 只是把这个事实写成了数据。

## 编码：共享部件 DAG

既然字与字之间共享部件，那就别重复存。构建阶段把所有字形组织成一个共享部件的 DAG：「氵」只存一次，几百个带三点水的字都引用它。引用深度限制在 8 层以内。GB 2312 展开需要 10858 个字形节点，把只被引用一次的节点内联掉之后，剩 **8243** 个。

坐标量化到 6 bit（区间 [-64, 256]——要留出界外余量，因为部件放置框可以把子部件推出画布再裁回来）。位流格式叫 FZE1，变长整数用指数哥伦布编码，好处是**不需要码表**：解码器不用先从 Flash 里载入一份 Huffman 表，省 RAM 也省启动逻辑。位流里每隔若干字放一个检查点，随机访问某个字不用从头解。检查点步长取 4 时总体积 127KB，取 8 时 124KB，我选了 4。

编码器是 OCaml 写的，跑在开发机上。整条流水线四步：解析 KAGE dump → 建共享部件 DAG → 单次引用内联 → 序列化成 FZE1 位流。下面按这个顺序把用到的代码全部贴出（ocamlformat 默认风格）。

### 解析 KAGE（kage.ml）

GlyphWiki dump 每行是 `名字|相关字|数据`；数据是 `$` 分隔的笔画，每个笔画是 `:` 分隔的整数，类型 99 表示部件引用：

```ocaml
(* KAGE / GlyphWiki data model.

   A glyph's `data` is `$`-separated strokes; each stroke is `:`-separated ints:
     type : headStyle : tailStyle : x1 : y1 : x2 : y2 [: ...]
   type 99 is a component reference:
     99 : a : b : x1 : y1 : x2 : y2 : glyphName [: stretch...]
   placing glyphName (defined on 0..200) into the box (x1,y1)-(x2,y2).

   Coordinates live on a 200x200 grid, origin top-left, y downward (like SVG).
   Real placement boxes solve the geometry that IDS topology could not. *)

type element =
  | Stroke of int * int array (* stroke type, flat coords x,y,x,y... *)
  | Ref of string * (int * int * int * int)
(* component name, box x1 y1 x2 y2 *)

type t = { data : (string, string) Hashtbl.t }

let geti f i =
  if i < Array.length f then try int_of_string (String.trim f.(i)) with _ -> 0
  else 0

let parse_data (s : string) : element list =
  String.split_on_char '$' s
  |> List.filter_map (fun seg ->
      let seg = String.trim seg in
      if seg = "" then None
      else begin
        let f = Array.of_list (String.split_on_char ':' seg) in
        let typ = geti f 0 in
        if typ = 99 then
          (* strip a KAGE variant/rotation suffix "name@N" -> base glyph "name" *)
          let raw = if Array.length f > 7 then String.trim f.(7) else "" in
          let name =
            match String.index_opt raw '@' with
            | Some i -> String.sub raw 0 i
            | None -> raw
          in
          if name = "" then None
          else Some (Ref (name, (geti f 3, geti f 4, geti f 5, geti f 6)))
        else begin
          let n = Array.length f in
          if n <= 3 then None
          else Some (Stroke (typ, Array.init (n - 3) (fun i -> geti f (i + 3))))
        end
      end)

(* Load the GlyphWiki dump text file (name | related | data). *)
let load path : t =
  let tbl = Hashtbl.create 900000 in
  let ic = open_in path in
  (try
     while true do
       let line = input_line ic in
       match String.split_on_char '|' line with
       | name :: _ :: data :: _ ->
           let name = String.trim name and data = String.trim data in
           if name <> "" && name <> "name" && String.contains data ':' then
             Hashtbl.replace tbl name data
       | _ -> ()
     done
   with End_of_file -> ());
  close_in ic;
  { data = tbl }

let mem t name = Hashtbl.mem t.data name
let get t name = Hashtbl.find_opt t.data name

(* Preferred GlyphWiki glyph name for a codepoint: prefer the PRC (-g) source
   shape, then the abstract entry, then other regional variants. *)
let name_candidates cp =
  let h = Printf.sprintf "u%04x" cp in
  [ h ^ "-g"; h; h ^ "-h"; h ^ "-t"; h ^ "-j"; h ^ "-k" ]

let glyph_name t cp = List.find_opt (fun n -> mem t n) (name_candidates cp)

let elements t name =
  match get t name with Some d -> parse_data d | None -> []
```

### 建共享部件 DAG（kcodec.ml）

坐标量化到 6 bit，然后把字符集递归展开成 DAG——部件名换成整数索引，同名部件天然只存一份：

```ocaml
(* ---- coordinate quantization over a fixed grid covering KAGE overhang ---- *)
let clo = -64.0
let chi = 256.0

let quant qbits v =
  let levels = (1 lsl qbits) - 1 in
  let i =
    int_of_float
      (Float.round
         ((float_of_int v -. clo) /. (chi -. clo) *. float_of_int levels))
  in
  if i < 0 then 0 else if i > levels then levels else i

(* ---- element model (resolved from KAGE, coords already absolute ints) ---- *)
type elem =
  | S of int * int array (* stroke type, coords x,y,... *)
  | R of int * int * int * int * int (* child glyph index, box x1 y1 x2 y2 *)

(* Build the DAG closure over a char list; returns:
   names array (index order), per-index element list (refs as indices), and
   char->index map for the requested codepoints. *)
let build_dag (kg : Kage.t) (cps : int list) =
  let idx = Hashtbl.create 40000 in
  (* name -> index *)
  let elems = Hashtbl.create 40000 in
  (* index -> elem list *)
  let count = ref 0 in
  let rec visit depth name =
    match Hashtbl.find_opt idx name with
    | Some i -> i
    | None ->
        let i = !count in
        Hashtbl.replace idx name i;
        incr count;
        let el =
          List.filter_map
            (function
              | Kage.Stroke (ty, c) -> Some (S (ty, c))
              | Kage.Ref (child, (x1, y1, x2, y2)) ->
                  if depth <= 8 && Kage.mem kg child then
                    Some (R (visit (depth + 1) child, x1, y1, x2, y2))
                  else None)
            (Kage.elements kg name)
        in
        Hashtbl.replace elems i el;
        i
  in
  let char_idx =
    List.map
      (fun cp ->
        match Kage.glyph_name kg cp with
        | Some n -> (cp, Some (visit 0 n))
        | None -> (cp, None))
      cps
  in
  let n = !count in
  let name_arr = Array.make n "" in
  Hashtbl.iter (fun name i -> name_arr.(i) <- name) idx;
  let elem_arr =
    Array.init n (fun i -> Option.value ~default:[] (Hashtbl.find_opt elems i))
  in
  (name_arr, elem_arr, char_idx)
```

### 序列化准备（kbin.ml）

位写入器、指数哥伦布原语，以及 `remapped_dag`——把字符重排到索引 0..6762（GB 2312 顺序，这样码点查字形就是行号查表），部件按被引用频率降序排在后面（频率高的 rank 小，指数哥伦布编出来就短）：

```ocaml
let step = 4

(* ---------- bit writer (MSB first) ---------- *)
type bw = { buf : Buffer.t; mutable cur : int; mutable n : int }

let bw_make () = { buf = Buffer.create (1 lsl 16); cur = 0; n = 0 }

let put_bit b v =
  b.cur <- (b.cur lsl 1) lor (v land 1);
  b.n <- b.n + 1;
  if b.n = 8 then (
    Buffer.add_char b.buf (Char.chr (b.cur land 0xFF));
    b.cur <- 0;
    b.n <- 0)

let put_bits b v k =
  for i = k - 1 downto 0 do
    put_bit b ((v lsr i) land 1)
  done

let bw_bitpos b = (Buffer.length b.buf * 8) + b.n

let bw_finish b =
  if b.n > 0 then (
    Buffer.add_char b.buf (Char.chr ((b.cur lsl (8 - b.n)) land 0xFF));
    b.cur <- 0;
    b.n <- 0);
  Buffer.contents b.buf

let nbits v =
  let k = ref 0 and x = ref v in
  while !x > 0 do
    incr k;
    x := !x lsr 1
  done;
  !k

(* Exp-Golomb order 0 for n >= 0 *)
let expg_put b n =
  let m = n + 1 in
  let k = nbits m in
  for _ = 1 to k - 1 do
    put_bit b 0
  done;
  put_bits b m k

let zigzag n = (n lsl 1) lxor (n asr 62)
let unzigzag z = (z lsr 1) lxor -(z land 1)

(* Exp-Golomb order k: low k bits raw, high part exp-Golomb(0). Table-free;
   k trades body density (small values) against tail length. *)
let expgk_put b n k =
  expg_put b (n lsr k);
  if k > 0 then put_bits b (n land ((1 lsl k) - 1)) k

type elem = S of int * int array | R of int * int * int * int * int

(* build DAG, remap so chars occupy indices 0..nchar-1 (GB2312 order). *)
let remapped_dag kg cps =
  let names, elems, char_idx = Kcodec.build_dag kg cps in
  let n = Array.length names in
  let newof = Array.make n (-1) in
  let counter = ref 0 in
  List.iter
    (fun (_cp, oi) ->
      match oi with
      | Some o when newof.(o) < 0 ->
          newof.(o) <- !counter;
          incr counter
      | _ -> ())
    char_idx;
  let nchar = !counter in
  (* remaining glyphs (components) ordered by descending reference frequency *)
  let freq = Array.make n 0 in
  Array.iter
    (List.iter (function
      | Kcodec.R (ci, _, _, _, _) -> freq.(ci) <- freq.(ci) + 1
      | _ -> ()))
    elems;
  let rest = ref [] in
  Array.iteri (fun o nv -> if nv < 0 then rest := o :: !rest) newof;
  let rest = List.sort (fun a b -> compare freq.(b) freq.(a)) !rest in
  List.iter
    (fun o ->
      newof.(o) <- !counter;
      incr counter)
    rest;
  (* rewrite elems into new-index order *)
  let out = Array.make n [] in
  Array.iteri
    (fun o els ->
      out.(newof.(o)) <-
        List.map
          (function
            | Kcodec.S (t, c) -> S (t, c)
            | Kcodec.R (ci, x1, y1, x2, y2) -> R (newof.(ci), x1, y1, x2, y2))
          els)
    elems;
  (out, nchar)
```

接着是**单次引用内联**：只被引用一次的部件没有共享价值，把它的元素直接拼进唯一的使用处（放置框逐层复合），DAG 从 10858 个节点缩到 8243 个：

```ocaml
(* Inline every component referenced exactly once (splice its elements into its
   unique use, composing placement boxes).  Removes single-use records + their
   refs without losing any sharing.  Chars (idx < nchar) are never removed. *)
let mapc lo hi v =
  lo
  + int_of_float
      (Float.round (float_of_int v /. 200. *. float_of_int (hi - lo)))

let xf (bx1, by1, bx2, by2) = function
  | S (ty, c) ->
      S
        ( ty,
          Array.mapi
            (fun i v -> if i land 1 = 0 then mapc bx1 bx2 v else mapc by1 by2 v)
            c )
  | R (k, kx1, ky1, kx2, ky2) ->
      R
        ( k,
          mapc bx1 bx2 kx1,
          mapc by1 by2 ky1,
          mapc bx1 bx2 kx2,
          mapc by1 by2 ky2 )

let simplify elems nchar =
  let cur = ref elems in
  let changed = ref true in
  while !changed do
    changed := false;
    let e = !cur in
    let rc = Array.make (Array.length e) 0 in
    Array.iter
      (List.iter (function
        | R (ci, _, _, _, _) -> rc.(ci) <- rc.(ci) + 1
        | S _ -> ()))
      e;
    cur :=
      Array.map
        (List.concat_map (function
          | S _ as s -> [ s ]
          | R (h, x1, y1, x2, y2) when h >= nchar && rc.(h) = 1 ->
              changed := true;
              List.map (xf (x1, y1, x2, y2)) e.(h)
          | R _ as r -> [ r ]))
        e
  done;
  (* compact: keep chars + still-referenced glyphs, remap indices *)
  let e = !cur in
  let rc = Array.make (Array.length e) 0 in
  Array.iter
    (List.iter (function
      | R (ci, _, _, _, _) -> rc.(ci) <- rc.(ci) + 1
      | S _ -> ()))
    e;
  let newof = Array.make (Array.length e) (-1) in
  let c = ref 0 in
  Array.iteri
    (fun i _ ->
      if i < nchar || rc.(i) > 0 then (
        newof.(i) <- !c;
        incr c))
    e;
  let out = Array.make !c [] in
  Array.iteri
    (fun i els ->
      if newof.(i) >= 0 then
        out.(newof.(i)) <-
          List.map
            (function
              | S _ as s -> s | R (k, a, b, cc, d) -> R (newof.(k), a, b, cc, d))
            els)
    e;
  out
```

### 编码位流（kbin.ml）

最后是位流编码的核心 `encode_expg`——每个字形记录是「元素个数 + 若干元素」，元素要么是笔画（首点绝对坐标 + 后续点 zig-zag 差分），要么是部件引用（部件 rank + 放置框在码本里的 rank）：

```ocaml
(* ---- exp-Golomb (no-Huffman) variant, for a speed/RAM comparison ---- *)
let encode_expg kg cps qbits =
  let elems0, nchar = remapped_dag kg cps in
  let elems = simplify elems0 nchar in
  let n = Array.length elems in
  let q v = Kcodec.quant qbits v in
  let boxf = Hashtbl.create 4000 and reff = Hashtbl.create 4000 in
  let bump t k =
    Hashtbl.replace t k (1 + Option.value ~default:0 (Hashtbl.find_opt t k))
  in
  Array.iter
    (List.iter (function
      | S _ -> ()
      | R (ci, x1, y1, x2, y2) ->
          bump reff ci;
          bump boxf (q x1, q y1, q x2, q y2)))
    elems;
  let ranked t =
    Hashtbl.fold (fun k c a -> (k, c) :: a) t []
    |> List.sort (fun (_, a) (_, b) -> compare b a)
    |> List.map fst
  in
  let box_alpha = Array.of_list (ranked boxf)
  and ref_alpha = Array.of_list (ranked reff) in
  let boxrank = Hashtbl.create 4000 and refrank = Hashtbl.create 4000 in
  Array.iteri (fun i b -> Hashtbl.replace boxrank b i) box_alpha;
  Array.iteri (fun i c -> Hashtbl.replace refrank c i) ref_alpha;
  let ib = nbits n in
  let rb = bw_make () and checkpoints = ref [] in
  Array.iteri
    (fun i els ->
      if i mod step = 0 then checkpoints := bw_bitpos rb :: !checkpoints;
      expg_put rb (List.length els);
      List.iter
        (function
          | S (_, c) ->
              let np = Array.length c / 2 in
              put_bit rb 0;
              expg_put rb np;
              let px = ref 0 and py = ref 0 in
              for i = 0 to np - 1 do
                let qx = q c.(2 * i) and qy = q c.((2 * i) + 1) in
                if i = 0 then (
                  put_bits rb qx qbits;
                  put_bits rb qy qbits)
                else (
                  expgk_put rb (zigzag (qx - !px)) 2;
                  expgk_put rb (zigzag (qy - !py)) 2);
                px := qx;
                py := qy
              done
          | R (ci, x1, y1, x2, y2) ->
              put_bit rb 1;
              expg_put rb (Hashtbl.find refrank ci);
              expg_put rb (Hashtbl.find boxrank (q x1, q y1, q x2, q y2)))
        els)
    elems;
  let recs = bw_finish rb in
  let checkpoints = Array.of_list (List.rev !checkpoints) in
  let hb = bw_make () in
  String.iter (fun ch -> put_bits hb (Char.code ch) 8) "FZE1";
  put_bits hb qbits 8;
  put_bits hb step 8;
  put_bits hb nchar 24;
  put_bits hb n 24;
  put_bits hb (Array.length box_alpha) 16;
  put_bits hb (Array.length ref_alpha) 24;
  put_bits hb (Array.length checkpoints) 24;
  Array.iter
    (fun (x1, y1, x2, y2) ->
      put_bits hb x1 qbits;
      put_bits hb y1 qbits;
      put_bits hb x2 qbits;
      put_bits hb y2 qbits)
    box_alpha;
  Array.iter (fun ci -> put_bits hb ci ib) ref_alpha;
  Array.iter (fun off -> put_bits hb off 24) checkpoints;
  bw_finish hb ^ recs
```

## 解码：不到 300 行，纯整数

解码器是一份 Rust，rustfmt 格式化后不到 300 行，`no_std` 友好，全程无浮点、无堆分配、无码表。给它一个码点和目标像素尺寸，它递归展开 DAG、变换坐标、吐出笔画折线，你拿去用 Bresenham 画就行。

```rust
// KAGE skeleton font decoder — exp-Golomb + int32 fixed-point + buffered CLZ
// bit reader + per-frame affine. no_std-friendly core (only main uses std).
// Format "FZE1" (see lib/kbin.ml encode_expg). f(index) -> [u8; 256].
// Build: rustc -O fzfont.rs && ./fzfont ../data/font_expg.bin 1946

const RES: usize = 16;
const FP: i32 = 512;

#[derive(Clone, Copy)]
struct Br {
    pos: usize,
    acc: u32,
    n: i32,
}
impl Br {
    fn refill(&mut self, f: &[u8]) {
        while self.n <= 24 {
            self.acc |= (f[self.pos] as u32) << (24 - self.n);
            self.pos += 1;
            self.n += 8;
        }
    }
    fn bits(&mut self, f: &[u8], k: i32) -> u32 {
        self.refill(f);
        let v = self.acc >> (32 - k);
        self.acc <<= k;
        self.n -= k;
        v
    }
    fn eg(&mut self, f: &[u8]) -> i64 {
        self.refill(f);
        let z = self.acc.leading_zeros() as i32;
        let t = 2 * z + 1;
        let m = self.acc >> (32 - t);
        self.acc <<= t;
        self.n -= t;
        m as i64 - 1
    }
    fn egk(&mut self, f: &[u8], k: i32) -> i64 {
        let hi = self.eg(f);
        (hi << k) | self.bits(f, k) as i64
    }
    fn tell(&self) -> i64 {
        self.pos as i64 * 8 - self.n as i64
    }
}
fn seek(f: &[u8], bit: i64) -> Br {
    let mut r = Br {
        pos: (bit >> 3) as usize,
        acc: 0,
        n: 0,
    };
    let o = (bit & 7) as i32;
    if o != 0 {
        r.refill(f);
        r.acc <<= o;
        r.n -= o;
    }
    r
}
fn unzig(z: i64) -> i32 {
    ((z >> 1) ^ -(z & 1)) as i32
}
fn nbits(mut x: i64) -> i32 {
    let mut k = 0;
    while x > 0 {
        k += 1;
        x >>= 1;
    }
    k
}

struct Font<'a> {
    b: &'a [u8],
    qb: i32,
    step: i64,
    ib: i32,
    box_off: i64,
    ref_off: i64,
    chk_off: i64,
    base: i64,
}
impl<'a> Font<'a> {
    fn new(b: &'a [u8]) -> Font<'a> {
        let mut r = seek(b, 0);
        r.bits(b, 16);
        r.bits(b, 16);
        let qb = r.bits(b, 8) as i32;
        let step = r.bits(b, 8) as i64;
        r.bits(b, 24);
        let n = r.bits(b, 24) as i64;
        let nbox = r.bits(b, 16) as i64;
        let nref = r.bits(b, 24) as i64;
        let nchk = r.bits(b, 24) as i64;
        let ib = nbits(n);
        let box_off = r.tell();
        let ref_off = box_off + nbox * 4 * qb as i64;
        let chk_off = ref_off + nref * ib as i64;
        let base = ((chk_off + nchk * 24) + 7) / 8 * 8;
        Font {
            b,
            qb,
            step,
            ib,
            box_off,
            ref_off,
            chk_off,
            base,
        }
    }
    fn refa(&self, k: i64) -> i64 {
        seek(self.b, self.ref_off + k * self.ib as i64).bits(self.b, self.ib) as i64
    }
    fn boxc(&self, k: i64) -> [i32; 4] {
        let mut r = seek(self.b, self.box_off + k * 4 * self.qb as i64);
        let mut o = [0i32; 4];
        for i in 0..4 {
            o[i] = r.bits(self.b, self.qb) as i32;
        }
        o
    }
    fn chk(&self, k: i64) -> i64 {
        seek(self.b, self.chk_off + k * 24).bits(self.b, 24) as i64
    }
}
fn dqf(q: i32) -> i32 {
    -64 * FP + q * 320 * FP / 63
}

fn plot(img: &mut [u8; RES * RES], x: i32, y: i32) {
    if (x as u32) < RES as u32 && (y as u32) < RES as u32 {
        img[y as usize * RES + x as usize] = 1;
    }
}
fn line(img: &mut [u8; RES * RES], mut x0: i32, mut y0: i32, x1: i32, y1: i32) {
    let dx = (x1 - x0).abs();
    let dy = (y1 - y0).abs();
    let sx = if x0 < x1 { 1 } else { -1 };
    let sy = if y0 < y1 { 1 } else { -1 };
    let mut e = dx - dy;
    loop {
        plot(img, x0, y0);
        if x0 == x1 && y0 == y1 {
            break;
        }
        let e2 = 2 * e;
        if e2 > -dy {
            e -= dy;
            x0 += sx;
        }
        if e2 < dx {
            e += dx;
            y0 += sy;
        }
    }
}
fn pix(s: i32) -> i32 {
    (s + FP / 2) >> 9
}

fn fskip(f: &Font, r: &mut Br) {
    let ne = r.eg(f.b);
    for _ in 0..ne {
        if r.bits(f.b, 1) == 0 {
            let np = r.eg(f.b);
            for i in 0..np {
                if i == 0 {
                    r.bits(f.b, f.qb);
                    r.bits(f.b, f.qb);
                } else {
                    r.egk(f.b, 2);
                    r.egk(f.b, 2);
                }
            }
        } else {
            r.eg(f.b);
            r.eg(f.b);
        }
    }
}

fn render(
    f: &Font,
    img: &mut [u8; RES * RES],
    g: i64,
    ox: i32,
    oy: i32,
    mx: i32,
    my: i32,
    depth: i32,
) {
    if depth > 8 {
        return;
    }
    let mut r = seek(f.b, f.base + f.chk(g / f.step));
    for _ in 0..(g % f.step) {
        fskip(f, &mut r);
    }
    let ax = (mx * 4480 * 8 + 6300) / 12600;
    let ay = (my * 4480 * 8 + 6300) / 12600;
    let bx = (ox - 64 * mx) * 14 / 200 + FP;
    let by = (oy - 64 * my) * 14 / 200 + FP;
    let ne = r.eg(f.b);
    for _ in 0..ne {
        if r.bits(f.b, 1) == 0 {
            let np = r.eg(f.b) as usize;
            let mut cs = [0i32; 32];
            let (mut px, mut py) = (0i32, 0i32);
            for i in 0..np {
                let (qx, qy) = if i == 0 {
                    (r.bits(f.b, f.qb) as i32, r.bits(f.b, f.qb) as i32)
                } else {
                    (px + unzig(r.egk(f.b, 2)), py + unzig(r.egk(f.b, 2)))
                };
                px = qx;
                py = qy;
                cs[2 * i] = ((ax * qx) >> 3) + bx;
                cs[2 * i + 1] = ((ay * qy) >> 3) + by;
            }
            if np == 2 {
                line(img, pix(cs[0]), pix(cs[1]), pix(cs[2]), pix(cs[3]));
            } else if np >= 3 {
                let (mut pxx, mut pyy) = (pix(cs[0]), pix(cs[1]));
                for s in 1..=5 {
                    let u = 5 - s;
                    let (bxx, byy);
                    if np == 3 {
                        bxx = (u * u * cs[0] + 2 * u * s * cs[2] + s * s * cs[4]) / 25;
                        byy = (u * u * cs[1] + 2 * u * s * cs[3] + s * s * cs[5]) / 25;
                    } else {
                        let j = if np - 1 > 3 { 3 } else { np - 1 };
                        bxx = (u * u * u * cs[0]
                            + 3 * u * u * s * cs[2]
                            + 3 * u * s * s * cs[4]
                            + s * s * s * cs[2 * j])
                            / 125;
                        byy = (u * u * u * cs[1]
                            + 3 * u * u * s * cs[3]
                            + 3 * u * s * s * cs[5]
                            + s * s * s * cs[2 * j + 1])
                            / 125;
                    }
                    let (cx, cy) = (pix(bxx), pix(byy));
                    line(img, pxx, pyy, cx, cy);
                    pxx = cx;
                    pyy = cy;
                }
            }
        } else {
            let rank = r.eg(f.b);
            let brank = r.eg(f.b);
            let child = f.refa(rank);
            let b = f.boxc(brank);
            let (b0, b1, b2, b3) = (dqf(b[0]), dqf(b[1]), dqf(b[2]), dqf(b[3]));
            let save = r;
            let nox = ox + ((b0 as i64 * mx as i64 / FP as i64) as i32);
            let noy = oy + ((b1 as i64 * my as i64 / FP as i64) as i32);
            let nmx = (mx as i64 * (b2 - b0) as i64 / (200 * FP) as i64) as i32;
            let nmy = (my as i64 * (b3 - b1) as i64 / (200 * FP) as i64) as i32;
            render(f, img, child, nox, noy, nmx, nmy, depth + 1);
            r = save;
        }
    }
}
fn glyph(f: &Font, i: i64) -> [u8; RES * RES] {
    let mut img = [0u8; RES * RES];
    render(f, &mut img, i, 0, 0, FP, FP, 0);
    img
}

fn main() {
    let a: Vec<String> = std::env::args().collect();
    let mut data = std::fs::read(&a[1]).unwrap();
    data.extend_from_slice(&[0; 8]);
    let f = Font::new(&data);
    let img = glyph(&f, a[2].parse().unwrap());
    for y in 0..RES {
        let mut s = String::new();
        for x in 0..RES {
            s.push(if img[y * RES + x] != 0 { '#' } else { '.' });
        }
        println!("{}", s);
    }
}
```

C 参考实现我也写了一份，单个头文件，逻辑和 Rust 版一一对应，方便直接塞进 MCU 工程：

```c
/* KAGE skeleton font decoder — exp-Golomb (table-free) + int32 fixed-point +
 * buffered CLZ bit reader + per-frame affine.  Pure f(glyph_index) -> 16x16.
 * No FPU used, no malloc, no Huffman tables. Set FONT to the font blob, call
 * init() once, then fz_glyph(index) which fills IMG[256] (1 byte/pixel).
 * Format: "FZE1", see lib/kbin.ml encode_expg. */
#include <stdint.h>
#include <string.h>

static const uint8_t* FONT;

/* buffered MSB-first bit reader (state = 12 bytes; cheap save/restore) */
typedef struct {
  long pos;
  uint32_t acc;
  int n;
} BR;
static inline void fz_refill(BR* r) {
  while (r->n <= 24) {
    r->acc |= (uint32_t)FONT[r->pos++] << (24 - r->n);
    r->n += 8;
  }
}
static inline uint32_t fz_bits(BR* r, int k) {
  fz_refill(r);
  uint32_t v = r->acc >> (32 - k);
  r->acc <<= k;
  r->n -= k;
  return v;
}
static void fz_seek(BR* r, long bit) {
  r->pos = bit >> 3;
  r->acc = 0;
  r->n = 0;
  int o = bit & 7;
  if (o) {
    fz_refill(r);
    r->acc <<= o;
    r->n -= o;
  }
}
static long fz_eg(BR* r) {
  fz_refill(r);
  int z = __builtin_clz(r->acc);
  int t = 2 * z + 1; /* CLZ-based exp-Golomb */
  uint32_t m = r->acc >> (32 - t);
  r->acc <<= t;
  r->n -= t;
  return (long)m - 1;
}
static long fz_egk(BR* r, int k) {
  long hi = fz_eg(r);
  return (hi << k) | fz_bits(r, k);
}
static long fz_unzig(long z) { return (z >> 1) ^ -(z & 1); }
static inline long fz_tell(BR* r) { return r->pos * 8 - r->n; }

static int QB, STEP, IB;
static long NBOX, NREF, BOX_OFF, REF_OFF, CHK_OFF, BASE;
static int fz_nb(long x) {
  int k = 0;
  while (x) {
    k++;
    x >>= 1;
  }
  return k;
}
static void init(void) {
  BR r;
  fz_seek(&r, 0);
  fz_bits(&r, 16);
  fz_bits(&r, 16);
  QB = fz_bits(&r, 8);
  STEP = fz_bits(&r, 8);
  fz_bits(&r, 24);
  long N = fz_bits(&r, 24);
  NBOX = fz_bits(&r, 16);
  NREF = fz_bits(&r, 24);
  long NCHK = fz_bits(&r, 24);
  IB = fz_nb(N);
  BOX_OFF = fz_tell(&r);
  REF_OFF = BOX_OFF + NBOX * 4 * QB;
  CHK_OFF = REF_OFF + NREF * IB;
  BASE = ((CHK_OFF + NCHK * 24) + 7) / 8 * 8;
}
static long fz_refa(long k) {
  BR r;
  fz_seek(&r, REF_OFF + k * IB);
  return fz_bits(&r, IB);
}
static void fz_box(long k, int* b) {
  BR r;
  fz_seek(&r, BOX_OFF + k * 4 * QB);
  for (int i = 0; i < 4; i++) b[i] = fz_bits(&r, QB);
}
static long fz_chk(long k) {
  BR r;
  fz_seek(&r, CHK_OFF + k * 24);
  return fz_bits(&r, 24);
}

/* fixed-point: FP=512, design coord = q dequantized; frames carry
   offset(design*FP) and scale(unit*FP). dqf only needed for placement boxes
   now. */
#define FP 512
typedef int32_t fx;
static fx dqf(int q) { return (fx)(-64) * FP + (fx)q * 320 * FP / 63; }

#ifndef RES /* output resolution; MCU default 16, override -DRES=N */
#define RES 16
#endif
static uint8_t IMG[RES * RES];
static void fz_plot(int x, int y) {
  if ((unsigned)x < RES && (unsigned)y < RES) IMG[y * RES + x] = 1;
}
static void fz_line(int x0, int y0, int x1, int y1) {
  int dx = x1 > x0 ? x1 - x0 : x0 - x1, dy = y1 > y0 ? y1 - y0 : y0 - y1,
      sx = x0 < x1 ? 1 : -1, sy = y0 < y1 ? 1 : -1, e = dx - dy;
  for (;;) {
    fz_plot(x0, y0);
    if (x0 == x1 && y0 == y1) break;
    int e2 = 2 * e;
    if (e2 > -dy) {
      e -= dy;
      x0 += sx;
    }
    if (e2 < dx) {
      e += dx;
      y0 += sy;
    }
  }
}
#define PIX(sfp) (((sfp) + FP / 2) >> 9)

static void fz_skip(BR* r) {
  int ne = fz_eg(r);
  for (int e = 0; e < ne; e++) {
    if (fz_bits(r, 1) == 0) {
      int np = fz_eg(r);
      for (int i = 0; i < np; i++) {
        if (i == 0) {
          fz_bits(r, QB);
          fz_bits(r, QB);
        } else {
          fz_egk(r, 2);
          fz_egk(r, 2);
        }
      }
    } else {
      fz_eg(r);
      fz_eg(r);
    }
  }
}

static void fz_render(long g, fx ox, fx oy, fx mx, fx my, int depth) {
  if (depth > 8) return;
  BR r;
  fz_seek(&r, BASE + fz_chk(g / STEP));
  for (long i = 0; i < g % STEP; i++) fz_skip(&r);
  /* per-frame affine: screen_fp = (A*q)>>3 + B (A has 3 extra fraction bits) */
  fx Ax = (mx * (320 * (RES - 2)) * 8 + 6300) / 12600,
     Ay = (my * (320 * (RES - 2)) * 8 + 6300) / 12600;
  fx Bx = (ox - 64 * mx) * (RES - 2) / 200 + FP,
     By = (oy - 64 * my) * (RES - 2) / 200 + FP;
  int ne = fz_eg(&r);
  for (int e = 0; e < ne; e++) {
    if (fz_bits(&r, 1) == 0) {
      int np = fz_eg(&r);
      int cs[32];
      int px = 0, py = 0;
      for (int i = 0; i < np; i++) {
        int qx, qy;
        if (i == 0) {
          qx = fz_bits(&r, QB);
          qy = fz_bits(&r, QB);
        } else {
          qx = px + (int)fz_unzig(fz_egk(&r, 2));
          qy = py + (int)fz_unzig(fz_egk(&r, 2));
        }
        px = qx;
        py = qy;
        cs[2 * i] = ((Ax * qx) >> 3) + Bx;
        cs[2 * i + 1] = ((Ay * qy) >> 3) + By;
      }
      if (np == 2)
        fz_line(PIX(cs[0]), PIX(cs[1]), PIX(cs[2]), PIX(cs[3]));
      else if (np >= 3) {
        int pxx = PIX(cs[0]), pyy = PIX(cs[1]);
        for (int s = 1; s <= 5; s++) {
          int u = 5 - s, bx, by;
          if (np == 3) {
            bx = (u * u * cs[0] + 2 * u * s * cs[2] + s * s * cs[4]) / 25;
            by = (u * u * cs[1] + 2 * u * s * cs[3] + s * s * cs[5]) / 25;
          } else {
            int j = np - 1 > 3 ? 3 : np - 1;
            bx = (u * u * u * cs[0] + 3 * u * u * s * cs[2] +
                  3 * u * s * s * cs[4] + s * s * s * cs[2 * j]) /
                 125;
            by = (u * u * u * cs[1] + 3 * u * u * s * cs[3] +
                  3 * u * s * s * cs[5] + s * s * s * cs[2 * j + 1]) /
                 125;
          }
          int cx = PIX(bx), cy = PIX(by);
          fz_line(pxx, pyy, cx, cy);
          pxx = cx;
          pyy = cy;
        }
      }
    } else {
      long rank = fz_eg(&r), brank = fz_eg(&r);
      long child = fz_refa(rank);
      int b[4];
      fz_box(brank, b);
      fx b0 = dqf(b[0]), b1 = dqf(b[1]), b2 = dqf(b[2]), b3 = dqf(b[3]);
      BR save = r;
      fx nox = ox + b0 * mx / FP, noy = oy + b1 * my / FP,
         nmx = mx * (b2 - b0) / (200 * FP), nmy = my * (b3 - b1) / (200 * FP);
      fz_render(child, nox, noy, nmx, nmy, depth + 1);
      r = save;
    }
  }
}
/* f(glyph_index) -> IMG[256] (1 byte/pixel) */
static void fz_glyph(long i) {
  memset(IMG, 0, sizeof IMG);
  fz_render(i, 0, 0, FP, FP, 0);
}
```

```c
/* Host tester for the FZE1 stroke-skeleton font decoder (fzfont.h).
 *   cc -O2 host.c -o host && ./host font_expg.bin 1946
 * Renders glyph index 1946 as ASCII; the index is a line number (0-based)
 * in gb2312.txt. */
#include <stdio.h>
#include <stdlib.h>

#include "fzfont.h"

int main(int argc, char** argv) {
  if (argc < 3) {
    fprintf(stderr, "usage: %s font.bin index\n", argv[0]);
    return 1;
  }
  FILE* f = fopen(argv[1], "rb");
  if (!f) {
    perror(argv[1]);
    return 1;
  }
  fseek(f, 0, SEEK_END);
  long sz = ftell(f);
  fseek(f, 0, SEEK_SET);
  uint8_t* buf = malloc(sz + 8);
  if (fread(buf, 1, sz, f) != (size_t)sz) {
    return 1;
  }
  fclose(f);
  memset(buf + sz, 0, 8); /* pad: buffered reader may over-read <=4 bytes */
  FONT = buf;
  init();
  fz_glyph(atol(argv[2]));
  for (int y = 0; y < RES; y++) {
    for (int x = 0; x < RES; x++) putchar(IMG[y * RES + x] ? '#' : '.');
    putchar('\n');
  }
  free(buf);
  return 0;
}
```

## 数据获取与复现

字形几何的唯一外部数据是 GlyphWiki 的公开 dump（约 115MB 压缩包，里面的 `dump_newest_only.txt` 就是 `Kage.load` 要读的文件）：

```sh
curl -fsSL -o dump.tar.gz https://glyphwiki.org/dump.tar.gz
tar -xzf dump.tar.gz dump_newest_only.txt
```

字符集表 `gb2312.txt` 是 6763 行十六进制 Unicode 码点，按 GB 2312 区位顺序排列（一级 3755 字 + 二级 3008 字），**字形索引就是行号**（0 起）。几行 Python 就能生成：

```python
out = []
for hi in range(0xB0, 0xF8):
    for lo in range(0xA1, 0xFF):
        try:
            out.append(f"{ord(bytes([hi, lo]).decode('gb2312')):04X}")
        except UnicodeDecodeError:
            pass
open("data/gb2312.txt", "w").write("\n".join(out) + "\n")  # 恰好 6763 行
```

OCaml 侧用 dune 组织，文中的模块按下面的布局摆放即可（`kbin.ml` 一节里的三段代码按顺序拼成一个文件）：

```
.
├── dune-project        # (lang dune 3.0)
├── lib
│   ├── dune            # (library (name fangzheng) (wrapped false))
│   ├── kage.ml
│   ├── kcodec.ml
│   └── kbin.ml
├── bin
│   ├── dune            # (executable (name main) (libraries fangzheng))
│   └── main.ml
└── data
    ├── dump_newest_only.txt
    └── gb2312.txt
```

然后一路跑通：

```sh
dune exec bin/main.exe                    # -> data/font_expg.bin（127KB）

# Rust 解码器：渲染「你」（gb2312.txt 第 1946 行，0 起）
rustc -O fzfont.rs && ./fzfont data/font_expg.bin 1946

# C 解码器，输出和 Rust 逐位一致
cc -O2 host.c -o host && ./host data/font_expg.bin 1946
```

编码是确定性的，位流逐字节可复现。C 和 Rust 解码器逻辑一致，输出也一致。
